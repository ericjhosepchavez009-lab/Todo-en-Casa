let pressTimer = null;
let isLongPress = false;

function setupLongPress() {
    document.querySelectorAll('.catalog-card').forEach(card => {
        const type = card.id.replace('card-', '');

        const start = (e) => {
            isLongPress = false;
            pressTimer = setTimeout(() => {
                isLongPress = true;
                openCategoryModal(type);
            }, 600);
        };

        const cancel = () => { clearTimeout(pressTimer); };

        card.addEventListener('mousedown', start);
        card.addEventListener('touchstart', start, {passive: true});
        card.addEventListener('mouseup', cancel);
        card.addEventListener('mouseleave', cancel);
        card.addEventListener('touchend', cancel);
    });
}

function handleCardClick(type, element) {
    if (isLongPress) return;
    openSection(type, element);
}

function toggleCartModal(show) {

function addToCart(item) {
    cart.push(item);

function removeFromCart(index) {
    cart.splice(index, 1);
    renderCart();
}

function renderCart() {
    const cartBody = document.getElementById('cartBody');
    const cartBadge = document.getElementById('cartBadge');
    const cartTotal = document.getElementById('cartTotal');

    cartBadge.innerText = cart.length;

    if (cart.length === 0) {
        cartBody.innerHTML = `
            <div class="cart-empty">
                <i class="fa-solid fa-basket-shopping"></i>
                <p>Tu carrito está vacío.<br>¡Añade algo del catálogo!</p>
            </div>
        `;
        cartTotal.innerText = "L 0";
        return;
    }

    let total = 0;
    cartBody.innerHTML = cart.map((item, index) => {
        total += item.price;
        return `
            <div class="cart-item">
                <div class="cart-item-info">
                    <h5>${item.title}</h5>
                    <small>${item.details}</small>
                    <div class="price">L ${item.price}</div>
                </div>
                <div class="cart-item-actions">
                    <button class="remove-btn" onclick="removeFromCart(${index})">
                        <i class="fa-solid fa-trash"></i>
                    </button>
                </div>
            </div>
        `;
    }).join('');

    cartTotal.innerText = `L ${total}`;
}

function sendWhatsAppOrder() {
    if (cart.length === 0) {
        alert("Tu carrito está vacío. Agrega productos antes de realizar tu pedido.");

    let message = "¡Hola! Vengo del catálogo web y me gustaría realizar el siguiente pedido:\n\n";

        message += `*${i + 1}. ${item.title}*\n`;
        message += `   • Detalle: ${item.details}\n`;
        message += `   • Precio: L ${item.price}\n\n`;
        grandTotal += item.price;
    });

    message += `------------------------------\n`;
    message += `*TOTAL A PAGAR: L ${grandTotal}*\n`;
    message += `------------------------------\n\n`;
    message += `Quedo a la espera de su confirmación. ¡Muchas gracias!`;

    window.open(`https://wa.me/${PHONE_NUMBER}?text=${encodeURIComponent(message)}`, "_blank");
}

function toggleDonutType() {
    const selectedRadio = document.querySelector('input[name="donutQty"]:checked');
    const type = selectedRadio.getAttribute('data-type');
    
    document.getElementById('simpleDonutsGroup').style.display = type === 'sencillas' ? 'block' : 'none';
    document.getElementById('toppingsDonutsGroup').style.display = type === 'bandeja' ? 'block' : 'none';
    
    updateDonutPrice();
}

function updateDonutPrice() {
    const selectedRadio = document.querySelector('input[name="donutQty"]:checked');
    const type = selectedRadio.getAttribute('data-type');
    
    if(type === 'sencillas') {
        let count = parseInt(document.getElementById('simpleDonutsCount').value) || 1;
        if(count < 1) count = 1;
        document.getElementById('donutTotalPrice').innerText = `Total: L ${count * 3}`;
        document.getElementById('donutToppingSummary').innerText = `${count} donitas sencillas (sin nada)`;
    } else {
        const basePrice = parseInt(selectedRadio.value);
        const checkboxes = document.querySelectorAll('.donut-topping:checked');
        const extraToppings = Math.max(0, checkboxes.length - 3);
        const totalPrice = basePrice + (extraToppings * 5);

        document.getElementById('donutTotalPrice').innerText = `Total: L ${totalPrice}`;
        document.getElementById('donutToppingSummary').innerText = `${checkboxes.length} toppings (${extraToppings > 0 ? `+L ${extraToppings * 5} extra` : 'sin recargo'})`;
    }
}

function addDonutsToCart() {
    const selectedRadio = document.querySelector('input[name="donutQty"]:checked');
    const type = selectedRadio.getAttribute('data-type');
    const name = selectedRadio.getAttribute('data-name');

    if(type === 'sencillas') {
        const count = parseInt(document.getElementById('simpleDonutsCount').value) || 1;
        addToCart({
            title: `Donitas sencillas (${count} unidades)`,
            details: "Sin toppings / tradicionales",
            price: count * 3
        });
    } else {
        const basePrice = parseInt(selectedRadio.value);
        const checkboxes = document.querySelectorAll('.donut-topping:checked');
        let toppings = Array.from(checkboxes).map(cb => cb.value);
        const extraToppings = Math.max(0, toppings.length - 3);
        const totalPrice = basePrice + (extraToppings * 5);

        addToCart({
            title: name,
            details: `Toppings: ${toppings.join(', ')}`,
            price: totalPrice
        });
    }
}

function addRolesToCart() {

    const baseLiquid = document.querySelector('input[name="roleBase"]:checked').value;
